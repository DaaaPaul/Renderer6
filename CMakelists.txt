cmake_minimum_required(VERSION 3.30)

project(Renderer6)

set(CMAKE_CXX_STANDARD 20)
set(CXX_STANDARD_REQUIRED ON)
set(CMAKE_MODULE_PATH ${CMAKE_CURRENT_SOURCE_DIR}/cmake)

find_package(Vulkan REQUIRED)
find_package(GLFW3 REQUIRED)
if(APPLE)
find_library(COCOA Cocoa)
find_library(IOKIT IOKit)
find_library(CORE_FOUNDATION CoreFoundation)
endif()
find_package(GLM REQUIRED)

add_executable(Renderer6)
target_sources(Renderer6 PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/src/Common.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/Common.h
    ${CMAKE_CURRENT_SOURCE_DIR}/src/GlfwWindowWrapper.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/GlfwWindowWrapper.hpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/main.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/RenderEngine.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/RenderEngine.h
    ${CMAKE_CURRENT_SOURCE_DIR}/src/TransformationMatrices.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/TransformationMatrices.hpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/Vertex.hpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/Vertex.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/VulkanBackendWrapper.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/VulkanBackendWrapper.hpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/VulkanDeviceLocalMemory.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/VulkanDeviceLocalMemory.hpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/VulkanDevicesWrapper.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/VulkanDevicesWrapper.hpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/VulkanGraphicsPipelineWrapper.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/VulkanGraphicsPipelineWrapper.hpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/VulkanHostVisibleMemory.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/VulkanHostVisibleMemory.hpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/VulkanMemoryCommon.h
    ${CMAKE_CURRENT_SOURCE_DIR}/src/VulkanMemoryCommon.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/VulkanPFNs.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/VulkanPFNs.h
    ${CMAKE_CURRENT_SOURCE_DIR}/src/VulkanSwapchainWrapper.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/VulkanSwapchainWrapper.hpp

    ${CMAKE_CURRENT_SOURCE_DIR}/shaders/shaders.slang
)
target_link_libraries(Renderer6 PRIVATE
    ${GLFW3_LIBRARY} 
    $<$<PLATFORM_ID:Darwin>:${COCOA}>
    $<$<PLATFORM_ID:Darwin>:${IOKIT}>
    $<$<PLATFORM_ID:Darwin>:${CORE_FOUNDATION}>
)
target_include_directories(Renderer6 PRIVATE
    ${GLFW3_INCLUDE_DIR}
    ${GLM_INCLUDE_DIRS}
    Vulkan::Headers
)
