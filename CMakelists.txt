cmake_minimum_required(VERSION 3.36)

project(Renderer6)

set(CMAKE_CXX_STANDARD 20)
set(CXX_STANDARD_REQUIRED ON)
set(CMAKE_MODULE_PATH ${CMAKE_CURRENT_SOURCE_DIR}/cmake)

find_package(Ktx REQUIRED)
find_package(Vulkan REQUIRED)
find_package(GLFW3 REQUIRED)
if(APPLE)
find_library(COCOA Cocoa)
find_library(IOKIT IOKit)
find_library(CORE_FOUNDATION CoreFoundation)
endif()
find_package(GLM REQUIRED)

set(DEBUG_MODE OFF)

add_executable(Renderer6)
set_property(DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR} PROPERTY VS_STARTUP_PROJECT Renderer6)
target_sources(Renderer6 PRIVATE
	${CMAKE_CURRENT_SOURCE_DIR}/src/MemoryDevice.cpp
	${CMAKE_CURRENT_SOURCE_DIR}/src/MemoryDevice.h
	${CMAKE_CURRENT_SOURCE_DIR}/src/MemoryHost.cpp
	${CMAKE_CURRENT_SOURCE_DIR}/src/MemoryHost.h
	${CMAKE_CURRENT_SOURCE_DIR}/src/Engine.cpp
	${CMAKE_CURRENT_SOURCE_DIR}/src/Engine.h
    ${CMAKE_CURRENT_SOURCE_DIR}/src/FeatureChain.hpp
	${CMAKE_CURRENT_SOURCE_DIR}/src/FrameData.cpp
	${CMAKE_CURRENT_SOURCE_DIR}/src/FrameData.h
	${CMAKE_CURRENT_SOURCE_DIR}/src/ImageViewHotspot.h
	${CMAKE_CURRENT_SOURCE_DIR}/src/ImageViewHotspot.cpp
	${CMAKE_CURRENT_SOURCE_DIR}/src/Instance.cpp
	${CMAKE_CURRENT_SOURCE_DIR}/src/Instance.h
	${CMAKE_CURRENT_SOURCE_DIR}/src/LogicalDevice.cpp
	${CMAKE_CURRENT_SOURCE_DIR}/src/LogicalDevice.h
	${CMAKE_CURRENT_SOURCE_DIR}/src/main.cpp
	${CMAKE_CURRENT_SOURCE_DIR}/src/Particle.cpp
	${CMAKE_CURRENT_SOURCE_DIR}/src/Particle.hpp
	${CMAKE_CURRENT_SOURCE_DIR}/src/PhysicalDevice.cpp
	${CMAKE_CURRENT_SOURCE_DIR}/src/PhysicalDevice.h
	${CMAKE_CURRENT_SOURCE_DIR}/src/PipelineLayouts.cpp
	${CMAKE_CURRENT_SOURCE_DIR}/src/PipelineLayouts.h
	${CMAKE_CURRENT_SOURCE_DIR}/src/Pipelines.cpp
	${CMAKE_CURRENT_SOURCE_DIR}/src/Pipelines.h
	${CMAKE_CURRENT_SOURCE_DIR}/src/Resources.cpp
	${CMAKE_CURRENT_SOURCE_DIR}/src/Resources.h
	${CMAKE_CURRENT_SOURCE_DIR}/src/ShaderModule.cpp
	${CMAKE_CURRENT_SOURCE_DIR}/src/ShaderModule.h
	${CMAKE_CURRENT_SOURCE_DIR}/src/Swapchain.cpp
	${CMAKE_CURRENT_SOURCE_DIR}/src/Swapchain.h
	${CMAKE_CURRENT_SOURCE_DIR}/src/Transforms.cpp
	${CMAKE_CURRENT_SOURCE_DIR}/src/Transforms.hpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/Util.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/Util.h
	${CMAKE_CURRENT_SOURCE_DIR}/src/Vertex.cpp
	${CMAKE_CURRENT_SOURCE_DIR}/src/Vertex.hpp
	${CMAKE_CURRENT_SOURCE_DIR}/src/Window.cpp
	${CMAKE_CURRENT_SOURCE_DIR}/src/Window.h

    ${CMAKE_CURRENT_SOURCE_DIR}/shaders/modelVertFrag.slang
	${CMAKE_CURRENT_SOURCE_DIR}/shaders/particleVertFrag.slang
	${CMAKE_CURRENT_SOURCE_DIR}/shaders/particleCompute.slang
    ${CMAKE_CURRENT_SOURCE_DIR}/TODO.txt
)
if(DEBUG_MODE)
	target_compile_options(Renderer6 PRIVATE
		$<$<CXX_COMPILER_ID:MSVC>:/W4>
		$<$<CXX_COMPILER_ID:MSVC>:/fsanitize=address /Zi>
	)
endif()
target_link_libraries(Renderer6 PRIVATE
    ${GLFW3_LIBRARY} 
	Vulkan::Vulkan
	KTX::ktx
    $<$<PLATFORM_ID:Darwin>:${COCOA}>
    $<$<PLATFORM_ID:Darwin>:${IOKIT}>
    $<$<PLATFORM_ID:Darwin>:${CORE_FOUNDATION}>
)
target_include_directories(Renderer6 PRIVATE
    ${GLFW3_INCLUDE_DIR}
    ${GLM_INCLUDE_DIRS}
	vendor
)
