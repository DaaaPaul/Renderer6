cmake_minimum_required(VERSION 3.30)

project(Renderer6)

set(CMAKE_CXX_STANDARD 20)
set(CXX_STANDARD_REQUIRED ON)
set(CMAKE_MODULE_PATH ${CMAKE_CURRENT_SOURCE_DIR}/cmake)

find_package(Ktx REQUIRED)
find_package(Vulkan REQUIRED)
find_package(GLFW3 REQUIRED)
if(APPLE)
find_library(COCOA Cocoa)
find_library(IOKIT IOKit)
find_library(CORE_FOUNDATION CoreFoundation)
endif()
find_package(GLM REQUIRED)

set(DEBUG_MODE OFF)

add_executable(Renderer6)
target_sources(Renderer6 PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/src/Common.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/Common.h
	${CMAKE_CURRENT_SOURCE_DIR}/src/DeviceLocal.cpp
	${CMAKE_CURRENT_SOURCE_DIR}/src/DeviceLocal.hpp
	${CMAKE_CURRENT_SOURCE_DIR}/src/DeviceMemory.cpp
	${CMAKE_CURRENT_SOURCE_DIR}/src/DeviceMemory.h
	${CMAKE_CURRENT_SOURCE_DIR}/src/Devices.cpp
	${CMAKE_CURRENT_SOURCE_DIR}/src/Devices.hpp
	${CMAKE_CURRENT_SOURCE_DIR}/src/Engine.cpp
	${CMAKE_CURRENT_SOURCE_DIR}/src/Engine.h
	${CMAKE_CURRENT_SOURCE_DIR}/src/GlobalState.cpp
	${CMAKE_CURRENT_SOURCE_DIR}/src/GlobalState.h
	${CMAKE_CURRENT_SOURCE_DIR}/src/GraphicsPipeline.cpp
	${CMAKE_CURRENT_SOURCE_DIR}/src/GraphicsPipeline.hpp
	${CMAKE_CURRENT_SOURCE_DIR}/src/HostVisible.cpp
	${CMAKE_CURRENT_SOURCE_DIR}/src/HostVisible.hpp
	${CMAKE_CURRENT_SOURCE_DIR}/src/Instance.cpp
	${CMAKE_CURRENT_SOURCE_DIR}/src/Instance.hpp
	${CMAKE_CURRENT_SOURCE_DIR}/src/main.cpp
	${CMAKE_CURRENT_SOURCE_DIR}/src/Swapchain.cpp
	${CMAKE_CURRENT_SOURCE_DIR}/src/Swapchain.hpp
	${CMAKE_CURRENT_SOURCE_DIR}/src/Transforms.cpp
	${CMAKE_CURRENT_SOURCE_DIR}/src/Transforms.hpp
	${CMAKE_CURRENT_SOURCE_DIR}/src/Vertex.cpp
	${CMAKE_CURRENT_SOURCE_DIR}/src/Vertex.hpp
	${CMAKE_CURRENT_SOURCE_DIR}/src/Window.cpp
	${CMAKE_CURRENT_SOURCE_DIR}/src/Window.hpp

    ${CMAKE_CURRENT_SOURCE_DIR}/shaders/shaders.slang
    ${CMAKE_CURRENT_SOURCE_DIR}/reminders.txt
)
if(DEBUG_MODE)
	target_compile_options(Renderer6 PRIVATE
		$<$<CXX_COMPILER_ID:MSVC>:/W4>
		$<$<CXX_COMPILER_ID:MSVC>:/fsanitize=address /Zi>
	)
endif()
target_link_libraries(Renderer6 PRIVATE
    ${GLFW3_LIBRARY} 
	Vulkan::Vulkan
	KTX::ktx
    $<$<PLATFORM_ID:Darwin>:${COCOA}>
    $<$<PLATFORM_ID:Darwin>:${IOKIT}>
    $<$<PLATFORM_ID:Darwin>:${CORE_FOUNDATION}>
)
target_include_directories(Renderer6 PRIVATE
    ${GLFW3_INCLUDE_DIR}
    ${GLM_INCLUDE_DIRS}
	vendor
)
