cmake_minimum_required(VERSION 3.30)

project(Renderer6)

set(CMAKE_CXX_STANDARD 20)
set(CXX_STANDARD_REQUIRED ON)
set(CMAKE_MODULE_PATH ${CMAKE_CURRENT_SOURCE_DIR}/cmake)

find_package(Vulkan REQUIRED)
find_package(GLFW3 REQUIRED)
if(APPLE)
find_library(COCOA Cocoa)
find_library(IOKIT IOKit)
find_library(CORE_FOUNDATION CoreFoundation)
endif()
find_package(GLM REQUIRED)

add_executable(Renderer6)
target_sources(Renderer6 PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/src/Device.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/DeviceLocalMemory.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/GraphicsPipeline.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/HostVisibleMemory.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/VulkanPFNs.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/main.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/RenderEngine.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/Swapchain.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/VulkanBackend.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/Window.cpp

    ${CMAKE_CURRENT_SOURCE_DIR}/src/Device.hpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/DeviceLocalMemory.hpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/GraphicsPipeline.hpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/HostVisibleMemory.hpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/VulkanPFNs.h
    ${CMAKE_CURRENT_SOURCE_DIR}/src/RenderEngine.hpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/Swapchain.hpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/VulkanBackend.hpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/Window.hpp

    ${CMAKE_CURRENT_SOURCE_DIR}/src/Common.h
    ${CMAKE_CURRENT_SOURCE_DIR}/src/GlobalParameters.h
)
target_link_libraries(Renderer6 PRIVATE
    ${GLFW3_LIBRARY} 
    $<$<PLATFORM_ID:Darwin>:${COCOA}>
    $<$<PLATFORM_ID:Darwin>:${IOKIT}>
    $<$<PLATFORM_ID:Darwin>:${CORE_FOUNDATION}>
)
target_include_directories(Renderer6 PRIVATE
    ${GLFW3_INCLUDE_DIR}
    ${GLM_INCLUDE_DIRS}
    Vulkan::Headers
)
if(WIN32)
target_compile_definitions(Renderer6 PRIVATE WINDOWS)
message(STATUS "Windows detected")
endif()
if(APPLE)
target_compile_definitions(Renderer6 PRIVATE MACOS)
message(STATUS "Mac detected")
endif()
